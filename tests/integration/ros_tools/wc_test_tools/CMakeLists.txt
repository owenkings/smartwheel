cmake_minimum_required(VERSION 3.16)
project(wc_test_tools)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(rmw REQUIRED)
add_executable(tf_ownership_probe src/tf_ownership_probe.cpp)
ament_target_dependencies(tf_ownership_probe rclcpp tf2_msgs rmw)
target_compile_options(tf_ownership_probe PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS tf_ownership_probe RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
