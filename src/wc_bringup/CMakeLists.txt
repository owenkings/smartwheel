cmake_minimum_required(VERSION 3.8)
project(wc_bringup)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_ogre_vendor REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets Network)
add_executable(mapping_rviz src/mapping_rviz.cpp src/mapping_teleop.cpp src/mapping_health.cpp)
target_include_directories(mapping_rviz PRIVATE include)
target_compile_features(mapping_rviz PRIVATE cxx_std_14)
target_link_libraries(mapping_rviz
  rclcpp::rclcpp
  rviz_common::rviz_common
  rviz_ogre_vendor::OgreMain
  rviz_ogre_vendor::OgreOverlay
  Qt5::Widgets Qt5::Network)
set_target_properties(mapping_rviz PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
install(TARGETS mapping_rviz DESTINATION lib/${PROJECT_NAME})
add_executable(manual_capture_ui src/manual_capture_ui.cpp src/mapping_teleop.cpp)
target_include_directories(manual_capture_ui PRIVATE include)
target_compile_features(manual_capture_ui PRIVATE cxx_std_14)
target_link_libraries(manual_capture_ui Qt5::Widgets Qt5::Network)
add_executable(map_save_choice src/map_save_choice.cpp)
target_compile_features(map_save_choice PRIVATE cxx_std_14)
target_link_libraries(map_save_choice Qt5::Widgets)
install(TARGETS manual_capture_ui map_save_choice DESTINATION lib/${PROJECT_NAME})
